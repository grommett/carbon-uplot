import { createHeatmap } from 'carbon-uplot';

const BUCKET_LABELS = ['0–10ms', '10–25ms', '25–50ms',
  '50–100ms', '100–250ms', '250–500ms', '500ms+'];

createHeatmap(el, {

  bucketLabels: BUCKET_LABELS,
});

/>
