import { createBarChart } from 'carbon-uplot';

function dayLabel(timestamp) {
  return new Date(timestamp * 1000).toLocaleDateString(undefined, {
    month: 'short',
    day: 'numeric',
  });
}

createBarChart(el, {
  data: [dailyTimestamps, dailyErrors],
  series: [{ label: 'Errors' }],
  xFormat: dayLabel,
  valueFormat: (value) => value.toLocaleString(),
});
